# Only the platform entropy/clock/mutex and C bucket store; not a full Archx build.
cmake_minimum_required(VERSION 3.20)
project(archx_platform_conformance LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(WIN32)
  set(platform_sources ../../src/platform_windows.c)
elseif(APPLE)
  set(platform_sources ../../src/platform_posix.c ../../src/platform_macos.c)
else()
  set(platform_sources ../../src/platform_posix.c ../../src/platform_linux.c)
endif()
add_executable(archx_platform_conformance ../test_platform.c ../../src/store.c ${platform_sources})
target_include_directories(archx_platform_conformance PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_platform_conformance PRIVATE /W4 /WX)
  target_link_libraries(archx_platform_conformance PRIVATE bcrypt)
else()
  target_compile_options(archx_platform_conformance PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(archx_platform_conformance PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
  if(APPLE)
    target_compile_definitions(archx_platform_conformance PRIVATE _DARWIN_C_SOURCE)
    find_library(SECURITY_FRAMEWORK Security REQUIRED)
    target_link_libraries(archx_platform_conformance PRIVATE ${SECURITY_FRAMEWORK})
  endif()
  find_package(Threads REQUIRED)
  target_link_libraries(archx_platform_conformance PRIVATE Threads::Threads m)
endif()
add_executable(archx_smbios_fixtures ../test_smbios.c ../../src/smbios.c)
target_include_directories(archx_smbios_fixtures PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_smbios_fixtures PRIVATE /W4 /WX)
else()
  target_compile_options(archx_smbios_fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME synthetic_smbios_fixtures COMMAND archx_smbios_fixtures)
add_test(NAME entropy_clock_mutex_store_conformance COMMAND archx_platform_conformance)

if(WIN32)
  add_executable(archx_windows_firmware ../test_windows_firmware.c ../../src/signals_windows.c ../../src/smbios.c)
  target_include_directories(archx_windows_firmware PRIVATE ../../src)
  if(MSVC)
    target_compile_options(archx_windows_firmware PRIVATE /W4 /WX)
  endif()
  add_test(NAME windows_firmware_provider COMMAND archx_windows_firmware)
endif()

add_executable(archx_uuid_fixtures ../test_uuid_value.c ../../src/uuid_value.c)
target_include_directories(archx_uuid_fixtures PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_uuid_fixtures PRIVATE /W4 /WX)
else()
  target_compile_options(archx_uuid_fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME synthetic_uuid_fixtures COMMAND archx_uuid_fixtures)
if(APPLE)
  find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
  add_executable(archx_macos_cf_uuid ../test_macos_cf_uuid.c ../../src/macos_cf_uuid.c ../../src/uuid_value.c)
  target_include_directories(archx_macos_cf_uuid PRIVATE ../../src)
  target_compile_options(archx_macos_cf_uuid PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(archx_macos_cf_uuid PRIVATE ${COREFOUNDATION_FRAMEWORK})
  add_test(NAME macos_synthetic_corefoundation_uuid COMMAND archx_macos_cf_uuid)
endif()
