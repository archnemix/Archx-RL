# Experimental primitives and actual core with a synthetic collector; not a product build.
cmake_minimum_required(VERSION 3.20)
project(archx_platform_conformance LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(WIN32)
  set(platform_sources ../../src/platform_windows.c)
elseif(APPLE)
  set(platform_sources ../../src/platform_posix.c ../../src/platform_macos.c)
else()
  set(platform_sources ../../src/platform_posix.c ../../src/platform_linux.c)
endif()
add_executable(archx_platform_conformance ../test_platform.c ../../src/store.c ${platform_sources})
target_include_directories(archx_platform_conformance PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_platform_conformance PRIVATE /W4 /WX)
  target_link_libraries(archx_platform_conformance PRIVATE bcrypt ws2_32)
else()
  target_compile_options(archx_platform_conformance PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(archx_platform_conformance PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
  if(APPLE)
    target_compile_definitions(archx_platform_conformance PRIVATE _DARWIN_C_SOURCE)
    find_library(SECURITY_FRAMEWORK Security REQUIRED)
    target_link_libraries(archx_platform_conformance PRIVATE ${SECURITY_FRAMEWORK})
  endif()
  find_package(Threads REQUIRED)
  target_link_libraries(archx_platform_conformance PRIVATE Threads::Threads m)
endif()
add_executable(archx_smbios_fixtures ../test_smbios.c ../../src/smbios.c)
target_include_directories(archx_smbios_fixtures PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_smbios_fixtures PRIVATE /W4 /WX)
else()
  target_compile_options(archx_smbios_fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME synthetic_smbios_fixtures COMMAND archx_smbios_fixtures)
add_test(NAME entropy_clock_mutex_store_conformance COMMAND archx_platform_conformance)
set_tests_properties(entropy_clock_mutex_store_conformance PROPERTIES TIMEOUT 30)

if(WIN32)
  add_executable(archx_windows_firmware ../test_windows_firmware.c ../../src/signals_windows.c ../../src/smbios.c)
  target_include_directories(archx_windows_firmware PRIVATE ../../src)
  if(MSVC)
    target_compile_options(archx_windows_firmware PRIVATE /W4 /WX)
  endif()
  add_test(NAME windows_firmware_provider COMMAND archx_windows_firmware)
endif()

add_executable(archx_uuid_fixtures ../test_uuid_value.c ../../src/uuid_value.c)
target_include_directories(archx_uuid_fixtures PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_uuid_fixtures PRIVATE /W4 /WX)
else()
  target_compile_options(archx_uuid_fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME synthetic_uuid_fixtures COMMAND archx_uuid_fixtures)
# Canonicalization is portable, independent of hardware collection.
add_executable(archx_canonical_fixtures ../test_canonical.c ../../src/signals.c)
target_include_directories(archx_canonical_fixtures PRIVATE ../../src)
if(MSVC)
  target_compile_options(archx_canonical_fixtures PRIVATE /W4 /WX)
else()
  target_compile_options(archx_canonical_fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME synthetic_canonical_fixtures COMMAND archx_canonical_fixtures)
if(APPLE)
  find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
  add_executable(archx_macos_cf_uuid ../test_macos_cf_uuid.c ../../src/macos_cf_uuid.c ../../src/uuid_value.c)
  target_include_directories(archx_macos_cf_uuid PRIVATE ../../src)
  target_compile_options(archx_macos_cf_uuid PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(archx_macos_cf_uuid PRIVATE ${COREFOUNDATION_FRAMEWORK})
  add_test(NAME macos_synthetic_corefoundation_uuid COMMAND archx_macos_cf_uuid)
endif()

# Crypto/parser fixtures are independently buildable; this is not a full core.
add_executable(archx_crypto_vectors ../test_vectors.c ../../src/sha256.c)
add_library(archx_crypto_fixture SHARED ../../src/sha256.c)
set_target_properties(archx_crypto_fixture PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(archx_json_contract ../test_json_contract.c ../../src/json.c)
foreach(target IN ITEMS archx_crypto_vectors archx_crypto_fixture archx_json_contract)
  target_include_directories(${target} PRIVATE ../../src)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(NOT MSVC)
  target_link_libraries(archx_json_contract PRIVATE m)
endif()
add_test(NAME synthetic_crypto_known_answer COMMAND archx_crypto_vectors)
add_test(NAME synthetic_json_contract COMMAND archx_json_contract)

# Actual common ABI linked to a synthetic Linux-shaped collector, solely for
# native core portability testing. This is not a supported non-Linux product.
add_library(archx_synthetic_core SHARED ../../src/archx.c ../../src/json.c
  ../../src/sha256.c ../../src/signals.c ../../src/store.c
  synthetic_collector.c ${platform_sources})
set_target_properties(archx_synthetic_core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(archx_synthetic_core PRIVATE ../../include ../../src)
add_executable(archx_synthetic_core_test test_core_synthetic.c)
target_include_directories(archx_synthetic_core_test PRIVATE ../../include)
target_link_libraries(archx_synthetic_core_test PRIVATE archx_synthetic_core)
if(MSVC)
  target_compile_options(archx_synthetic_core PRIVATE /W4 /WX)
  target_compile_options(archx_synthetic_core_test PRIVATE /W4 /WX)
  target_link_libraries(archx_synthetic_core PRIVATE bcrypt ws2_32)
else()
  foreach(target IN ITEMS archx_synthetic_core archx_synthetic_core_test)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
  endforeach()
  target_link_libraries(archx_synthetic_core PRIVATE Threads::Threads m)
  if(APPLE)
    target_compile_definitions(archx_synthetic_core PRIVATE _DARWIN_C_SOURCE)
    target_link_libraries(archx_synthetic_core PRIVATE ${SECURITY_FRAMEWORK})
  endif()
endif()
add_test(NAME synthetic_actual_core_abi COMMAND archx_synthetic_core_test)
