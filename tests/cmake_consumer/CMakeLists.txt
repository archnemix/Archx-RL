cmake_minimum_required(VERSION 3.20)
project(archx_consumer LANGUAGES C)
find_package(Archx CONFIG REQUIRED)
if(NOT Archx_ABI_VERSION STREQUAL "1")
  message(FATAL_ERROR "Unexpected Archx ABI version")
endif()
foreach(kind IN ITEMS core static)
  add_executable(consumer_${kind} consumer.c)
  target_link_libraries(consumer_${kind} PRIVATE Archx::${kind})
endforeach()
enable_testing()
add_test(NAME shared_consumer COMMAND consumer_core)
add_test(NAME static_consumer COMMAND consumer_static)
