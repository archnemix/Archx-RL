cmake_minimum_required(VERSION 3.20)
project(archx_native_permission_survey LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(WIN32)
  add_executable(native_permission_survey native_windows.c ../../src/signals_windows.c ../../src/smbios.c)
  target_include_directories(native_permission_survey PRIVATE ../../src)
  target_link_libraries(native_permission_survey PRIVATE advapi32)
  if(MSVC)
    target_compile_options(native_permission_survey PRIVATE /W4 /WX)
  endif()
elseif(APPLE)
  add_executable(native_permission_survey native_macos.c)
  target_compile_options(native_permission_survey PRIVATE -Wall -Wextra -Wpedantic -Werror)
  find_library(IOKIT_FRAMEWORK IOKit REQUIRED)
  find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
  target_link_libraries(native_permission_survey PRIVATE ${IOKIT_FRAMEWORK} ${COREFOUNDATION_FRAMEWORK})
else()
  message(FATAL_ERROR "This survey supports Windows and macOS only")
endif()
